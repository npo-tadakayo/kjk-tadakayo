### #1 領収書のメール送付で編集UIがPDFに写る — `admin/js/supply-print.js`

**原因**: `html2pdf().from(document.getElementById("body"))` は画面のDOMをそのまま描く。
`@media print` の `.rcpt-noprint{display:none}` は html2canvas に効かないため、
用途区分のプルダウン・×ボタン・「行を追加」・入力枠がそのまま入る。

**方針**: PDF化の直前に `#body` を複製し、印刷時と同じ見え方に整えた複製を渡す。

```js
// 画面の #body を「印刷と同じ見え方」に整えた複製にして返す。
// html2canvas は @media print を適用しないので、.rcpt-noprint や <input> が写ってしまう。
// 複製側で input/select を確定値の文字に置き換え、編集専用の要素を取り除く。
function printableClone(){
  const src = document.getElementById("body");
  const clone = src.cloneNode(true);
  // 入力欄は「今の値」を文字にする。cloneNode は属性しか写さないので元要素から読む
  const srcIn = src.querySelectorAll("input, select, textarea");
  const dstIn = clone.querySelectorAll("input, select, textarea");
  dstIn.forEach((el, i) => {
    const o = srcIn[i];
    const span = document.createElement("span");
    span.textContent = o.tagName === "SELECT" ? (o.options[o.selectedIndex]?.text || "") : o.value;
    if (el.classList.contains("num")) span.className = "num";
    el.replaceWith(span);
  });
  clone.querySelectorAll(".rcpt-noprint").forEach((n) => n.remove());
  // html2canvas は描画済みのDOMが要るので、画面外に置いて渡す
  const holder = document.createElement("div");
  holder.style.cssText = `position:fixed;left:-10000px;top:0;width:${src.offsetWidth}px;background:#fff`;
  holder.appendChild(clone);
  document.body.appendChild(holder);
  return { clone, cleanup: () => holder.remove() };
}
```

`setupMailDoc` の送信処理を次のように差し替える:

```js
- const el = document.getElementById("body");
- const opt = { ... };
- const dataUri = await window.html2pdf().set(opt).from(el).outputPdf("datauristring");
+ const { clone, cleanup } = printableClone();
+ let dataUri;
+ try {
+   const opt = { ... };   // 既存のまま
+   dataUri = await window.html2pdf().set(opt).from(clone).outputPdf("datauristring");
+ } finally { cleanup(); }
```

**検証**: 領収書ページで送信直前の複製を `document.body.appendChild` したまま停止し、
`clone.querySelectorAll("input,select,.rcpt-noprint").length === 0` を確認。請求書側は複製しても内容が変わらないことを確認。

### #3 出荷修正の在庫差分をトランザクションにする — `admin/js/supply.js: saveShipEdit`

**原因**: 差分の基準が「モーダルを開いた時点の items」。保存時に読み直していない。
かつ shipments → products → inventoryMovements が別々の書き込みで、途中失敗で食い違う。

```js
const btn=document.getElementById("saveShipBtn"); btn.disabled=true;
try{
  const shipRef = doc(db,"shipments",s0._id);
  let moved = 0;
  await runTransaction(db, async (tx)=>{
    const snap = await tx.get(shipRef);
    if(!snap.exists()) throw new Error("この出荷は削除されています。画面を更新してください");
    const cur = snap.data();
    // 開いたあとに誰かが直していたら止める（黙って上書きしない）
    const a=cur.updatedAt?.toMillis?.()||0, b=s0.updatedAt?.toMillis?.()||0;
    if(a!==b) throw new Error("他の人が先にこの出荷を変更しています。画面を更新してから直してください");

    let delta={};
    if(items){
      // 差分は「今の保存内容」から計算する（開いた時点の s0.items ではなく）
      const before=qtyBySku(cur.items), after=qtyBySku(items);
      Object.keys({...before,...after}).forEach(sku=>{ delta[sku]=(after[sku]||0)-(before[sku]||0); });
      if(shipUsedStock(cur)){
        // 読み取りは書き込みより前にまとめて済ませる（Firestore トランザクションの決まり）
        for(const sku of Object.keys(delta)){
          if(delta[sku]<=0) continue;
          const p=(await tx.get(doc(db,"products",sku))).data()||{};
          if((p.stock||0)<delta[sku]) throw new Error(`在庫不足: ${p.name||sku}（在庫 ${p.stock||0} / 追加で必要 ${delta[sku]}）`);
        }
      }
    }
    tx.update(shipRef, patch);
    if(items && shipUsedStock(cur)){
      for(const sku of Object.keys(delta)){
        const d=delta[sku]; if(!d) continue;
        tx.update(doc(db,"products",sku),{stock:increment(-d)});
        tx.set(doc(collection(db,"inventoryMovements")),{ sku, delta:-d, reason:"shipment_edit",
          refNo:s0.soNumber, createdAt:serverTimestamp(), userName:currentUser.displayName||currentUser.email });
        moved++;
      }
    }
  });
  document.getElementById("shipModal").classList.remove("open");
  editingShip=null;
  toast(locked ? `出荷 ${s0.soNumber} の送付先を直しました`
               : `出荷 ${s0.soNumber} を修正しました${moved?"（在庫も差分を調整）":""}`);
}catch(e){ alert(`修正の保存に失敗しました: ${e.message}`); }
finally{ btn.disabled=false; }
```

保存前の在庫チェック（`products` 配列を見る既存ブロック）は残してよいが、最終判定はトランザクション内が正。
`runTransaction` / `increment` は supply.js に import 済み。
